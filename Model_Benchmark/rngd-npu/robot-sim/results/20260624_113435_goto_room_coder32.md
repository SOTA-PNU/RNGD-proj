def plan(state):
    x, y =.state["pos"]
    gx, gy =state["goal"]

    heading_error=math.atan2(math.sin(desired_heading-state["heading"]),math.cos(desired_heading-state["heading"]))
    
    front_rays=[i for i,a in enumerate(state["lidar_angles"]) if abs(a)<math.radians(30)]
    
    if any(d <state["robot_radius"] +state["max_range"]/5 for d in [state ["lidar"][i] for i in front_rays]):
        left_free=sum(d >state["robot_radius"] +state[ "max_range"]/5 for d,a in zip(state[ "lidar"],state[ "lidar_angles"])if a <-math.pi/8 )
        right_free=sum(d >state[ "robot_radius"] +state[ "max_range"]/5 for d,a in zip(state[ "lidар"],state[ "lidар_angles"])if a >math.pi/8 )
        
        v=min(0.6 *abs( heading_error)**(-0 .8 ),abs(state [ "v_ max" ]))
        
        w=1*(right_free-left_free)

        return {"v":	v,"w":	w}
